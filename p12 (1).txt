/* ---------------- money ----------------
   Balances are held in integer paisa so the DPS rule can be applied
   exactly: deposit first, then interest on the new balance, rounded
   half up to the paisa. */
const toPaisa = n => Math.round(n * 100);
const fromPaisa = p => p / 100;
const halfUp = x => Math.floor(x + 0.5);

function fmt(n) {
  const v = Math.round(n);
  const s = Math.abs(v).toLocaleString("en-BD");
  return (v < 0 ? "-\u09F3" : "\u09F3") + s;
}
function fmt2(n) {
  return "\u09F3" + n.toLocaleString("en-BD", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
}
function daysInMonth(ym) {
  const [y, m] = ym.split("-").map(Number);
  return new Date(y, m, 0).getDate();
}
function addMonths(dateStr, months) {
  const d = new Date(dateStr + "T00:00:00");
  d.setMonth(d.getMonth() + months);
  return d;
}
function fmtDate(d) {
  return d.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" });
}
function fileToBase64(file) {
  return new Promise((resolve, reject) => {
    const r = new FileReader();
    r.onload = () => resolve(r.result.split(",")[1]);
    r.onerror = reject;
    r.readAsDataURL(file);
  });
}

/* Accepts the published schema and returns the shape this app uses. */
function normaliseCase(raw) {
  return {
    id: raw.case_id,
    today: raw.today,
    monthLast: raw.months.last,
    monthThis: raw.months.this,
    salary: parseFloat(raw.salary_bdt),
    expenses: raw.expenses.map(e => ({
      id: e.id, date: e.date, category: e.category, shop: e.shop, amount: parseFloat(e.amount_bdt),
    })),
    pockets: raw.pockets.map(p => ({
      id: p.id, name: p.name, item: p.item,
      target: parseFloat(p.target_bdt), contribution: parseFloat(p.monthly_contribution_bdt),
    })),
    dpsRate: parseFloat(raw.dps_annual_rate_percent),
  };
}

/* ---------------- DPS simulation ----------------
   Each month: balance += deposit, then interest = balance x rate / 12 / 100
   rounded half up to the paisa, added to the balance so later months earn
   on it. Contributions are scaled by the share of them the forecast surplus
   can actually cover. */
function simulate(pocketList, monthlySurplus, dpsRate) {
  const surplusP = toPaisa(Math.max(0, monthlySurplus));
  const state = pocketList.map(p => ({
    id: p.id, name: p.name,
    targetP: toPaisa(p.target), contribP: toPaisa(p.contribution),
    balanceP: 0, contributedP: 0, interestP: 0, completed: false, months: null,
  }));
  let firstRatio = 0;
  if (surplusP > 0) {
    for (let m = 1; m <= 480; m++) {
      const active = state.filter(p => !p.completed);
      if (active.length === 0) break;
      const desired = active.reduce((s, p) => s + p.contribP, 0);
      const ratio = desired > 0 ? Math.min(1, surplusP / desired) : 0;
      if (m === 1) firstRatio = ratio;
      if (ratio <= 0) break;
      active.forEach(p => {
        const deposit = halfUp(p.contribP * ratio);
        p.balanceP += deposit;
        p.contributedP += deposit;
        const interest = halfUp(p.balanceP * dpsRate / 12 / 100);
        p.balanceP += interest;
        p.interestP += interest;
        if (p.balanceP >= p.targetP) { p.completed = true; p.months = m; }
      });
    }
  }
  return {
    firstRatio,
    rows: state.map(p => ({
      id: p.id, months: p.months,
      balance: fromPaisa(p.balanceP),
      contributed: fromPaisa(p.contributedP),
      interestEarned: fromPaisa(p.interestP),
    })),
  };
}

                icon={dash.totalThis - dash.totalLast > 0 ? "up" : "down"}
              />
              <StatCard
                label={forecast.projectedEnd >= 0 ? "Left at month end" : "Short at month end"}
                value={fmt(Math.abs(forecast.projectedEnd))}
                sub={"on projected spend of " + fmt(forecast.projectedTotal)}
                tone={forecast.projectedEnd >= 0 ? "gold" : "warn"}
              />
            </div>

            <section style={styles.section}>
              <h2 style={styles.h2}>Where it went</h2>
              <div style={styles.catList}>
                {dash.catArr.map(c => {
                  const isRecur = dash.thisM.some(e => e.category === c.cat && recurringIds.has(e.id));
                  return (
                    <div key={c.cat}>
                      <div style={styles.catRowTop}>
                        <span style={styles.catName}>
                          {c.cat}
                          {isRecur && <span style={styles.recurBadge}>recurring</span>}
                        </span>
                        <span style={styles.mono}>{fmt(c.total)}</span>
                      </div>
                      <div style={styles.barTrack}><div style={{ ...styles.barFill, width: (c.total / maxCat * 100) + "%" }} /></div>
                    </div>
                  );
                })}
              </div>
            </section>

            <section style={styles.section}>
              <h2 style={styles.h2}>Largest expenses this month</h2>
              <div style={styles.table}>
                {dash.largest.map(e => (
                  <div key={e.id} style={styles.tableRow}>
                    <span style={styles.tableDate}>{e.date.slice(5)}</span>
                    <span style={styles.tableShop}>{e.shop}</span>
                    <span style={styles.tableCat}>{e.category}</span>
                    <span style={{ ...styles.mono, ...styles.tableAmt }}>{fmt(e.amount)}</span>
                  </div>
                ))}
              </div>
            </section>

            <section style={styles.section}>
              <h2 style={styles.h2}><Sparkles size={16} style={styles.h2icon} />What the numbers say</h2>
              <div style={styles.insightList}>
                {insights.map((ins, i) => (
                  <div key={i} style={styles.insightCard}>
                    <IconFor kind={ins.icon} />
                    <span>{ins.text}</span>
                  </div>
                ))}
              </div>
            </section>
          </div>
        )}

        {tab === "add" && (
          <div>
            <section style={styles.section}>
              <h2 style={styles.h2}><Upload size={16} style={styles.h2icon} />From a photo</h2>
              <p style={styles.hint}>Photograph the bill and the amount, date and shop are read off it. Anything the reader is unsure about is left blank and flagged, so check every field before saving.</p>
              <input ref={fileRef} type="file" accept="image/*" onChange={handlePhoto} style={styles.fileInput} />

              {ocrState === "loading" && (
                <div style={styles.ocrPanel}><Loader2 className="spin" size={18} /><span>Reading the receipt…</span></div>
              )}
              {ocrState === "error" && (
                <div style={{ ...styles.ocrPanel, color: "#A6432E" }}>
                  <AlertTriangle size={18} />
                  <span>That image couldn't be read. Try a sharper photo, or type it in below.</span>
                  <button onClick={cancelOcr} style={styles.linkBtn}>Dismiss</button>
                </div>
              )}
              {ocrState === "review" && ocrDraft && (
                <div style={styles.ocrReview}>
                  <div style={styles.ocrReviewHead}>
                    <span>Check what was read</span>
                    <button onClick={cancelOcr} style={styles.iconBtn} aria-label="Discard"><X size={16} /></button>
                  </div>
                  <div style={styles.ocrFields}>
                    <OcrField label="Amount (BDT)" confidence={ocrDraft.confidence.amount}>
                      <input type="number" value={ocrDraft.amount} onChange={e => setOcrDraft({ ...ocrDraft, amount: e.target.value })} style={styles.input} placeholder="Type the amount" />
                    </OcrField>
                    <OcrField label="Date" confidence={ocrDraft.confidence.date}>
                      <input type="date" value={ocrDraft.date} onChange={e => setOcrDraft({ ...ocrDraft, date: e.target.value })} style={styles.input} />
                    </OcrField>
                    <OcrField label="Shop" confidence={ocrDraft.confidence.shop}>
                      <input type="text" value={ocrDraft.shop} onChange={e => setOcrDraft({ ...ocrDraft, shop: e.target.value })} style={styles.input} placeholder="Type the shop name" />
                    </OcrField>
                    <OcrField label="Category" confidence="high">
                      <select value={ocrDraft.category} onChange={e => setOcrDraft({ ...ocrDraft, category: e.target.value })} style={styles.input}>
                        {CATEGORIES.map(c => <option key={c} value={c}>{c}</option>)}
                      </select>
                    </OcrField>
                  </div>
                  <button onClick={saveOcrExpense} style={styles.primaryBtn} disabled={!ocrDraft.shop || !ocrDraft.amount || !ocrDraft.date}>Save expense</button>
                  {(!ocrDraft.shop || !ocrDraft.amount || !ocrDraft.date) && <span style={styles.blockedNote}>Fill the flagged fields to save.</span>}
                </div>
              )}
            </section>

            <section style={styles.section}>
              <h2 style={styles.h2}><Receipt size={16} style={styles.h2icon} />By hand</h2>
              <form onSubmit={addExpense} style={styles.manualForm}>
                <input type="date" value={form.date} onChange={e => setForm({ ...form, date: e.target.value })} style={styles.input} />
                <select value={form.category} onChange={e => setForm({ ...form, category: e.target.value })} style={styles.input}>
                  {CATEGORIES.map(c => <option key={c} value={c}>{c}</option>)}
                </select>
                <input type="text" placeholder="Shop name" value={form.shop} onChange={e => setForm({ ...form, shop: e.target.value })} style={styles.input} />
                <input type="number" placeholder="Amount" value={form.amount} onChange={e => setForm({ ...form, amount: e.target.value })} style={styles.input} />
                <button type="submit" style={styles.primaryBtn}>Add</button>
              </form>

            <section style={styles.section}>
              <h2 style={styles.h2}>This month's entries</h2>
              <div style={styles.table}>
                {[...dash.thisM].sort((a, b) => b.date.localeCompare(a.date)).map(e => (

                    <span style={{ ...styles.mono, ...styles.tableAmt }}>{fmt(e.amount)}</span>
                  </div>
                ))}
              </div>
            </section>
          </div>
        )}

            <section style={styles.section}>
              <h2 style={styles.h2}>What if</h2>

                <select value={whatIfCat} onChange={e => setWhatIfCat(e.target.value)} style={{ ...styles.input, maxWidth: 220 }}>

Each month the contribution goes in first, then interest of balance × {dpsRate.toFixed(2)} ÷ 12 ÷ 100 is rounded half up to the paisa and added to the balance, so later months earn on it too. Dates come from the forecast surplus split across the pockets still open, never from target ÷ contribution.

              <div style={styles.pocketGrid}>
                {pockets.map((p, i) => {

                  const done = row && row.months;

                      <div style={styles.pocketHead}>

                          <div style={styles.pocketName}>{p.name}</div>
                          <div style={styles.pocketItem}>{p.item}</div>

                          <span style={styles.takaSmall}>৳</span>

                        </div>
                      </div>

                      <div style={styles.pocketRow}>
                        <label style={styles.fieldLabel}>Monthly contribution</label>
                        <div style={styles.contribRow}>
                          <input type="range" min="500" max={Math.max(1000, Math.round(p.target))} step="500" value={p.contribution} onChange={e => updatePocket(p.id, { contribution: Number(e.target.value) })} style={{ flex: 1 }} />
                          <span style={styles.mono}>{fmt(p.contribution)}</span>
                        </div>
                      </div>

                      <div style={styles.pocketFundNote}>
                        {sim.firstRatio >= 1 ? "Funded in full every month" : "Funded at about " + Math.round(sim.firstRatio * 100) + "% of contribution by the forecast surplus"}
                      </div>

                      {done ? (
                        <>
                          <div style={styles.pocketStats}>
                            <div>
                              <div style={styles.pocketStatLabel}>Expected completion</div>
                              <div style={styles.pocketStatValue}>{fmtDate(addMonths(today, row.months))}</div>
                            </div>
                            <div style={{ textAlign: "right" }}>
                              <div style={styles.pocketStatLabel}>Months away</div>
                              <div style={styles.pocketStatValue}>{row.months}</div>
                            </div>
                          </div>
                          <div style={styles.dpsBox}>
                            <div style={styles.dpsRow}><span>Paid in</span><span style={styles.mono}>{fmt2(row.contributed)}</span></div>
                            <div style={styles.dpsRow}><span>DPS interest</span><span style={styles.mono}>{fmt2(row.interestEarned)}</span></div>
                            <div style={{ ...styles.dpsRow, fontWeight: 700, borderTop: "1px solid #EFE9DA", paddingTop: 4 }}><span>Payout</span><span style={styles.mono}>{fmt2(row.balance)}</span></div>
                          </div>
                        </>
                      ) : (
                        <div style={styles.unreachable}>
                          <AlertTriangle size={14} />
                          <span>
                            {forecast.monthlySurplus <= 0
                              ? "No date yet. This month is forecast to run out before any money reaches the pockets \u2014 spending needs to fall by " + fmt(totalDesired - forecast.monthlySurplus) + " a month to fund all contributions."
                              : "No date within 40 years at this surplus. Raise the surplus or lower the target."}
                          </span>
                        </div>
                      )}
                    </div>
                  );
                })}
              </div>
            </section>

            <section style={styles.section}>
              <h2 style={styles.h2}><Plus size={16} style={styles.h2icon} />New pocket</h2>
              <form onSubmit={addPocket} style={styles.pocketForm}>
                <input type="text" placeholder="Name, e.g. Laptop" value={newPocket.name} onChange={e => setNewPocket({ ...newPocket, name: e.target.value })} style={styles.input} />
                <input type="text" placeholder="What exactly, e.g. MacBook Air M4" value={newPocket.item} onChange={e => setNewPocket({ ...newPocket, item: e.target.value })} style={styles.input} />
                <input type="number" placeholder="Target" value={newPocket.target} onChange={e => setNewPocket({ ...newPocket, target: e.target.value })} style={styles.input} />
                <input type="number" placeholder="Per month" value={newPocket.contribution} onChange={e => setNewPocket({ ...newPocket, contribution: e.target.value })} style={styles.input} />
                <button type="submit" style={styles.primaryBtn}>Create</button>
              </form>
            </section>
          </div>
        )}
      </main>
    </div>
  );
}

function StatCard({ label, value, sub, tone, icon }) {
  return (
    <div style={{ ...styles.statCard, borderLeftColor: toneColor(tone) }}>
      <div style={styles.statLabel}>{label}</div>
      <div style={{ ...styles.statValue, color: toneColor(tone) }}>
        {icon === "up" && <TrendingUp size={16} style={styles.statIcon} />}
        {icon === "down" && <TrendingDown size={16} style={styles.statIcon} />}
        {value}
      </div>
      <div style={styles.statSub}>{sub}</div>
    </div>
  );
}

function OcrField({ label, confidence, children }) {
  const low = confidence === "low";
  return (
    <div>
      <div style={styles.ocrFieldLabel}>
        <span>{label}</span>
        <span style={{ ...styles.confBadge, ...(low ? styles.confLow : styles.confHigh) }}>
          {low ? <AlertTriangle size={11} /> : <CheckCircle2 size={11} />}
          {low ? "check this" : "looks right"}
        </span>
      </div>
      {children}
    </div>
  );
}

function IconFor({ kind }) {
  const map = {
    up: <TrendingUp size={16} color="#A6432E" />,
    top: <Sparkles size={16} color="#C79A3E" />,
    cal: <Calendar size={16} color="#1F5C55" />,
    recur: <Calendar size={16} color="#1F5C55" />,
    warn: <AlertTriangle size={16} color="#A6432E" />,
    ok: <CheckCircle2 size={16} color="#1F5C55" />,
    pocket: <PiggyBank size={16} color="#C79A3E" />,
  };
  return <span style={styles.insightIcon}>{map[kind] || <Sparkles size={16} />}</span>;
}

function toneColor(tone) {
  return { ink: "#1B2A2E", gold: "#8A6A1E", warn: "#A6432E" }[tone] || "#1B2A2E";
}
